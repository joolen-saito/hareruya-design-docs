```php
    #[Route('/%eccube_api_v1_route%/admin/optionBulkPurchaseId.json', name: 'api_admin_option_bulk_purchase_id', methods: ['GET'])]
    public function getBulkPurchaseId(): JsonResponse
    {
        $Option = $this->mtbOptionRepository->findOneBy([
            'option_key' => MtbOption::BULK_PURCHASE_ID,
        ]);

        $value = $Option?->getOptionValue() ?? '';

        return new JsonResponse($value);
    }
```
